cmake_minimum_required(VERSION 3.20)
project(MapSpooner LANGUAGES CXX)

option(MAP_SPOONER_BUILD_ASI "Build GTA V's ASI module" ON)
set(SCRIPTHOOKV_SDK "" CACHE PATH "Directory containing the ScriptHookV SDK inc and lib folders")
set(MAP_SPOONER_LEGACY_ADAPTER "" CACHE FILEPATH "Optional verified Legacy adapter C++ source")

if(MAP_SPOONER_BUILD_ASI)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "MapSpooner.asi requires a Windows x64 toolchain.")
    endif()
    find_path(SHV_INCLUDE main.h PATHS "${SCRIPTHOOKV_SDK}/inc" NO_DEFAULT_PATH REQUIRED)
    find_library(SHV_LIBRARY ScriptHookV PATHS "${SCRIPTHOOKV_SDK}/lib" NO_DEFAULT_PATH REQUIRED)

    if(MAP_SPOONER_LEGACY_ADAPTER)
        set(adapter "${MAP_SPOONER_LEGACY_ADAPTER}")
    else()
        set(adapter src/Legacy.cpp)
    endif()

    add_library(MapSpooner SHARED src/Script.cpp src/Pattern.cpp "${adapter}")
    target_include_directories(MapSpooner PRIVATE src "${SHV_INCLUDE}")
    target_link_libraries(MapSpooner PRIVATE "${SHV_LIBRARY}" user32)
    target_compile_features(MapSpooner PRIVATE cxx_std_20)
    target_compile_definitions(MapSpooner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(MapSpooner PROPERTIES PREFIX "" SUFFIX ".asi")
    if(MSVC)
        target_compile_options(MapSpooner PRIVATE /W4 /permissive-)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(MapSpoonerCoreTest tests/Core.cpp)
    target_include_directories(MapSpoonerCoreTest PRIVATE src)
    target_compile_features(MapSpoonerCoreTest PRIVATE cxx_std_20)
    add_test(NAME MapSpoonerCoreTest COMMAND MapSpoonerCoreTest)
endif()
